@app.route("/register", methods=["POST"])
def register():

    try:

        data = request.get_json()

        auth_service.register(
            data["username"],
            data["password"]
        )

        return jsonify({
            "message": "User registered successfully"
        }), 201


    except ValidationError as e:

        return jsonify({
            "error": str(e)
        }), 400


    except Exception:

        return jsonify({
            "error": "Internal Server Error"
        }), 500
